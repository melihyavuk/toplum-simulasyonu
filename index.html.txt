<script>
    // Global Değişkenler ve Veri Seti
    let historyData = [];
    let currentMonth = 0;
    let antigravityActive = false;

    // Gerçekçi Toplum Verileri (Segmentasyon)
    const segments = [
        { name: "Düşük Gelir", count: 5000, avgIncome: 17002, mpc: 0.95 },
        { name: "Orta Gelir", count: 3500, avgIncome: 45000, mpc: 0.75 },
        { name: "Yüksek Gelir", count: 1500, avgIncome: 130000, mpc: 0.45 }
    ];

    // Grafik Kurulumu
    const ctx = document.getElementById('spendingChart').getContext('2d');
    const chart = new Chart(ctx, {
        type: 'line',
        data: {
            labels: [],
            datasets: [{
                label: 'Aylık Toplam Harcama (Milyon ₺)',
                data: [],
                borderColor: '#1e40af',
                backgroundColor: 'rgba(30, 64, 175, 0.1)',
                fill: true,
                tension: 0.4
            }]
        },
        options: {
            responsive: true,
            plugins: { legend: { display: true } },
            scales: { y: { beginAtZero: false, title: { display: true, text: 'Milyon ₺' } } }
        }
    });

    // Slider Değerlerini Dinleme
    const inputs = ['taxRate', 'inflation', 'confidence', 'interest'];
    inputs.forEach(id => {
        const el = document.getElementById(id);
        const valSpan = document.getElementById(id.replace('Rate', '').replace('ion', '').replace('ence', '').replace('est', '') + 'Val');
        el.oninput = () => {
            valSpan.innerText = (id === 'confidence' ? '' : '%') + el.value;
        };
    });

    // Ana Simülasyon Fonksiyonu
    function nextMonth() {
        const tax = antigravityActive ? 0 : document.getElementById('taxRate').value / 100;
        const inf = antigravityActive ? 0 : document.getElementById('inflation').value / 100;
        const conf = antigravityActive ? 1.5 : document.getElementById('confidence').value / 100;
        const intr = antigravityActive ? 0 : document.getElementById('interest').value / 100;

        let monthlyTotalSpend = 0;

        segments.forEach(s => {
            let disposableIncome = s.avgIncome * (1 - tax);
            // Formül: Temel Harcama + (Gelir * MPC * Güven) - (Enflasyon ve Faiz Baskısı)
            let spendPerPerson = (disposableIncome * s.mpc * conf) / (1 + (intr * 0.3) + (inf * 0.1));
            
            // Antigravity modunda harcama ivmelenir
            if(antigravityActive) spendPerPerson *= 1.8;

            monthlyTotalSpend += spendPerPerson * s.count;
        });

        currentMonth++;
        const spendInMillions = (monthlyTotalSpend / 1000000).toFixed(2);

        // Veri Kaydı
        historyData.push({
            ay: currentMonth,
            harcama: spendInMillions,
            vergi: tax,
            enflasyon: inf,
            guven: conf,
            faiz: intr
        });

        // UI Güncelleme
        document.getElementById('displaySpend').innerText = spendInMillions + "M ₺";
        document.getElementById('displayMonth').innerText = currentMonth;

        // Grafik Güncelleme
        chart.data.labels.push("Ay " + currentMonth);
        chart.data.datasets[0].data.push(spendInMillions);
        chart.update();
    }

    // ANALİZ İÇİN CSV DIŞA AKTARMA
    function exportCSV() {
        if(historyData.length === 0) return alert("Önce simülasyonu ilerletmelisiniz!");
        
        let csvContent = "data:text/csv;charset=utf-8,Ay,Harcama(Milyon TL),Vergi Orani,Enflasyon,Guven Endeksi,Faiz Orani\n";
        
        historyData.forEach(row => {
            csvContent += `${row.ay},${row.harcama},${row.vergi},${row.enflasyon},${row.guven},${row.faiz}\n`;
        });

        const encodedUri = encodeURI(csvContent);
        const link = document.createElement("a");
        link.setAttribute("href", encodedUri);
        link.setAttribute("download", "toplum_harcama_analizi.csv");
        document.body.appendChild(link);
        link.click();
    }

    // SIHIRLI ANTIGRAVITY FONKSIYONU
    window.antigravity = function() {
        antigravityActive = !antigravityActive;
        document.getElementById('antigravity-label').style.display = antigravityActive ? 'block' : 'none';
        console.log("Antigravity modu " + (antigravityActive ? "aktif!" : "devre dışı."));
        return "Yerçekimi ayarları değiştirildi.";
    };

    // Başlangıçta bir veri noktası oluştur
    nextMonth();
</script>
